## 5. 复现步骤

```bash
cd ~/repos/projects/duplex-voice/tools/ruler1
DATA=~/.hermes/workspace/ruler1/data

# 1) 下载（断点续传，跳过已完成）
python3 fetch.py BiggestWave/ChorusDuplex-SideTalk $DATA/chorus --jobs 6
python3 fetch.py zlinao/WearVox $DATA/wearvox --jobs 6
python3 fetch.py ASLP-lab/HumDial-FDBench $DATA/humdial_dl --jobs 1
unzip -oq $DATA/humdial_dl/Humdial-Track2-Test.zip "test/cn_test_nondev/*" -d $DATA/humdial

# 2) 生成清单
python3 build_manifests.py --dataset chorus  --data $DATA/chorus       --out manifests
python3 build_manifests.py --dataset wearvox --data $DATA/wearvox      --out manifests
python3 build_manifests.py --dataset humdial --data $DATA/humdial      --out manifests

# 3) 基线校验 + 判分
python3 baselines.py --manifest manifests/chorus.jsonl --mode silent  --out predictions/silent__chorus.jsonl
python3 score.py --manifest manifests/chorus.jsonl --pred predictions/silent__chorus.jsonl --sweep
```
